```{r}
fit_eid <- if(file.exists("data/fit_eid.rds")) readRDS("data/fit_eid.rds") else NULL
```

## Ringkasan Fit Model (CFA)

```{r}
#| label: tbl-summary-cfa
#| tbl-cap: "Tabel Ringkasan Fit Model CFA Keseluruhan"
df_cfa <- rbind(
  if(!is.null(fit_eid)) get_cfa_fit_df(fit_eid, "EID") else NULL,
  if(!is.null(fit_esi)) get_cfa_fit_df(fit_esi, "ESI") else NULL,
  if(!is.null(fit_psmls)) get_cfa_fit_df(fit_psmls, "PSMLS") else NULL,
  if(!is.null(fit_peb)) get_cfa_fit_df(fit_peb, "PEB") else NULL,
  if(!is.null(fit_pwb)) get_cfa_fit_df(fit_pwb, "PWB") else NULL
)

if (!is.null(df_cfa) && nrow(df_cfa) > 0) {
  knitr::kable(df_cfa)

}
```

```{r}
# Menonaktifkan autentikasi Google

df[] <- lapply(df, function(x) {
  if (is.list(x)) {
    x <- unlist(lapply(x, function(v) if(is.null(v)) NA else as.character(v)))
  }
  return(x)
})
df <- as.data.frame(df, check.names = FALSE)

# Mapping nama variabel demografi agar mudah dianalisis (Menggunakan pencarian kata kunci agar anti-error)
```

```{r}
print_demo_table <- function(variabel, labels) {
    freq <- table(variabel, useNA = "ifany")
    pct  <- prop.table(freq) * 100
    display <- sapply(names(freq), function(x) {
        if (is.na(x)) return("Missing")
        if (x %in% names(labels)) return(labels[[x]])
        return(x)

    data.frame(
        Kategori   = unname(display),
        Frekuensi  = as.numeric(freq),
        Persentase = paste0(round(pct, 2), "%"),
```

# Model Pengukuran Final {#sec-cfa}

## CFA Environmental Identity (EID) {#sec-cfa-eid}

Model pengukuran unidimensional untuk EID (14 aitem). *Dioptimasi dengan toleransi 3 Modification Indices terbesar.*

```{r}
#| label: cfa-eid
model_eid <- "
  EID =~ EID_1 + EID_2 + EID_3 + EID_4 + EID_5 + EID_6 + EID_7 + EID_8 + EID_9 + EID_10 + EID_11 + EID_12 + EID_13 + EID_14 
  EID_7 ~~ EID_12 
  EID_3 ~~ EID_6 
  EID_11 ~~ EID_13 
"
fit_eid <- cfa(model_eid, data = df, std.lv = TRUE)
summary(fit_eid, fit.measures = TRUE, standardized = TRUE)
```

### Reliabilitas EID

```{r}
#| label: tbl-rel-eid

.alpha_eid <- compRelSEM(fit_eid, tau.eq = TRUE)
```

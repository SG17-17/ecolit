Pure unidimensional measurement model for EID (14 items).

```{r}
#| label: setup-cfa-eid
#| include: false

source("R/helpers.R")
df <- readRDS("data/cleaned_data.rds")
```

```{r}
#| label: cfa-eid
model_eid <- "
  EID =~ EID_1 + EID_2 + EID_3 + EID_4 + EID_5 + EID_6 + EID_7 + EID_8 + EID_9 + EID_10 + EID_11 + EID_12 + EID_13 + EID_14 
"
fit_eid <- cfa(model_eid, data = df, std.lv = TRUE, estimator = "MLR")
summary(fit_eid, fit.measures = TRUE, standardized = TRUE)

# Save results for summary page
saveRDS(fit_eid, "data/fit_eid.rds")
```

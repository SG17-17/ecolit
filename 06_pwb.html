## Definisi Model

```{r}
#| label: cfa-pwb
model_pwb <- "
  Autonomy =~ PWB_1 + PWB_7 + PWB_13_R + PWB_19_R + PWB_25 + PWB_31_R + PWB_37
  Mastery =~ PWB_2 + PWB_8 + PWB_14_R + PWB_20 + PWB_26_R + PWB_32_R + PWB_38
  Growth =~ PWB_3_R + PWB_9 + PWB_15_R + PWB_21 + PWB_27_R + PWB_33 + PWB_39_R
  Relations =~ PWB_4 + PWB_10_R + PWB_16_R + PWB_22 + PWB_28 + PWB_34_R + PWB_40
  Purpose =~ PWB_5_R + PWB_11 + PWB_17_R + PWB_23_R + PWB_29 + PWB_35 + PWB_41_R
  Acceptance =~ PWB_6 + PWB_12 + PWB_18_R + PWB_24 + PWB_30_R + PWB_36_R + PWB_42
  PWB_34_R ~~ PWB_30_R 
  PWB_26_R ~~ PWB_22 
  PWB_37 ~~ PWB_36_R 
"
fit_pwb_clean <- tryCatch(cfa(model_pwb, data = df_clean, std.lv = TRUE), error = function(e) NULL)
fit_pwb_raw   <- tryCatch(cfa(model_pwb, data = df_raw, std.lv = TRUE), error = function(e) NULL)

if(!is.null(fit_pwb_clean)) saveRDS(fit_pwb_clean, "data/fit_pwb_clean.rds")
if(!is.null(fit_pwb_raw)) saveRDS(fit_pwb_raw, "data/fit_pwb_raw.rds")
```
